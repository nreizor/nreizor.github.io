<script>
const canvas = document.getElementById("board");
const ctx = canvas.getContext("2d");

const scoreText = document.getElementById("score");
const bestText = document.getElementById("best");
const startBtn = document.getElementById("start");

const cell = 20;
const count = 20;

let snake;
let food;
let dir;
let nextDir;
let score = 0;
let best = 0;
let timer = null;
let running = false;
let ended = false;

try {
  best = Number(localStorage.getItem("snakeBest")) || 0;
} catch (e) {}

bestText.textContent = best;

function reset() {
  clearInterval(timer);
  timer = null;

  snake = [
    {x: 10, y: 10},
    {x: 9, y: 10},
    {x: 8, y: 10}
  ];

  dir = {x: 1, y: 0};
  nextDir = {x: 1, y: 0};

  score = 0;
  running = false;
  ended = false;

  scoreText.textContent = score;
  startBtn.textContent = "Начать";

  spawnFood();
  draw();
}

function spawnFood() {
  do {
    food = {
      x: Math.floor(Math.random() * count),
      y: Math.floor(Math.random() * count)
    };
  } while (
    snake.some(p => p.x === food.x && p.y === food.y)
  );
}

function draw() {
  ctx.fillStyle = "#080f20";
  ctx.fillRect(0, 0, 400, 400);

  // Игровая сетка
  ctx.strokeStyle = "#172440";

  for (let i = 0; i <= count; i++) {
    ctx.beginPath();
    ctx.moveTo(i * cell, 0);
    ctx.lineTo(i * cell, 400);
    ctx.stroke();

    ctx.beginPath();
    ctx.moveTo(0, i * cell);
    ctx.lineTo(400, i * cell);
    ctx.stroke();
  }

  // Еда
  ctx.fillStyle = "#ff6584";
  ctx.beginPath();
  ctx.arc(
    food.x * cell + 10,
    food.y * cell + 10,
    7,
    0,
    Math.PI * 2
  );
  ctx.fill();

  //


ейка
  snake.forEach(function(p, i) {
    ctx.fillStyle = i === 0 ? "#73fbd3" : "#35c9a7";

    ctx.beginPath();
    ctx.rect(
      p.x * cell + 2,
      p.y * cell + 2,
      cell - 4,
      cell - 4
    );
    ctx.fill();
  });

  if (!running) {
    ctx.fillStyle = "rgba(0,0,0,0.65)";
    ctx.fillRect(0, 0, 400, 400);

    ctx.textAlign = "center";
    ctx.fillStyle = "white";
    ctx.font = "bold 25px Arial";

    ctx.fillText(
      ended ? "Игра окончена!" : "Готов играть?",
      200,
      190
    );

    ctx.font = "16px Arial";
    ctx.fillText(
      ended ? "Нажми «Заново»" : "Нажми «Начать»",
      200,
      220
    );
  }
}

function tick() {
  dir = nextDir;

  const head = {
    x: snake[0].x + dir.x,
    y: snake[0].y + dir.y
  };

  const ate = head.x === food.x && head.y === food.y;

  const body = ate ? snake : snake.slice(0, -1);

  const wall =
    head.x < 0 ||
    head.y < 0 ||
    head.x >= count ||
    head.y >= count;

  const self = body.some(function(p) {
    return p.x === head.x && p.y === head.y;
  });

  if (wall || self) {
    running = false;
    ended = true;
    clearInterval(timer);
    timer = null;
    draw();
    return;
  }

  snake.unshift(head);

  if (ate) {
    score += 10;
    scoreText.textContent = score;

    if (score > best) {
      best = score;
      bestText.textContent = best;

      try {
        localStorage.setItem("snakeBest", String(best));
      } catch (e) {}
    }

    spawnFood();
  } else {
    snake.pop();
  }

  draw();
}

function start() {
  if (running) return;

  if (ended) reset();

  running = true;
  startBtn.textContent = "Продолжить";

  clearInterval(timer);

  timer = setInterval(
    tick,
    Math.max(65, 145 - score / 2)
  );

  draw();
}

function pause() {
  if (!running) return;

  running = false;
  clearInterval(timer);
  timer = null;

  draw();
}

function changeDirection(name) {
  const directions = {
    up: {x: 0, y: -1},
    down: {x: 0, y: 1},
    left: {x: -1, y: 0},
    right: {x: 1, y: 0}
  };

  const d = directions[name];

  if (!d) return;

  if (d.x === -dir.x && d.y === -dir.y) {
    return;
  }

  nextDir = d;
}

// Кнопки на телефоне
document.querySelectorAll("[data-dir]").forEach(function(btn) {
  btn.addEventListener("click", function() {
    changeDirection(btn.dataset.dir);

    if (!running && !ended) {
      start();
    }
  });
});

// Запуск
startBtn.addEventListener("click", start);

// Пауза
document.getElementById("pause").addEventListener(
  "click",
  pause
);

// Перезапуск
document.getElementById("restart").addEventListener(
  "click",
  reset
);

// Управление клавиатурой
document.addEventListener("keydown", function(e) {
  const keys = {
    ArrowUp: "up",
    ArrowDown: "down",
    ArrowLeft: "left",
    ArrowRight: "right",
    w: "up",
    W: "up",
    s: "down",
    S: "down",
    a: "left",
    A: "left",
    d: "right",
    D: "right"
  };

  if (keys[e.key]) {
    e.preventDefault();
    changeDirection(keys[e.key]);

    if (!running && !ended) {
      start();
    }
  }

  if (e.code === "Space") {
    e.preventDefault();

    if (running) {
      pause();
    } else {
      start();
    }
  }
});

// Свайпы на телефоне
let touchX = 0;
let touchY = 0;

canvas.addEventListener("touchstart", function(e) {
  touchX = e.changedTouches[0].clientX;
  touchY = e.changedTouches[0].clientY;
}, {passive: true});

canvas.addEventListener("touchend", function(e) {
  const dx = e.changedTouches[0].clientX - touchX;
  const dy = e.changedTouches[0].clientY - touchY;

  if (Math.max(Math.abs(dx), Math.abs(dy)) < 20) {
    return;
  }

  if (Math.abs(dx) > Math.abs(dy)) {
    changeDirection(dx > 0 ? "right" : "left");
  } else {
    changeDirection(dy > 0 ? "down" : "up");
  }

  if (!running && !ended) {
    start();
  }
}, {passive: true});

reset();
</script>
